class Validator:
    def __init__(self, min_value: float = 0.0, max_value: float = 1_000_000.0):
        self.min_value = min_value
        self.max_value = max_value

    def validate(self, record: Record) -> list[str]:
        errors: list[str] = []
        if not record.record_id:
            errors.append("record_id is required")
        if not record.name:
            errors.append("name is required")
        if not record.category:
            errors.append("category is required")
        if not isinstance(record.value, (int, float)):
            errors.append("value must be numeric")
        elif not self.min_value <= record.value <= self.max_value:
            errors.append("value is outside the accepted range")
        if len(record.name) > 200:
            errors.append("name is too long")
        if len(record.tags) > 50:
            errors.append("too many tags")
        return errors

    def validate_or_raise(self, record: Record) -> Record:
        errors = self.validate(record)
        if errors:
            raise ValidationError("; ".join(errors))
        return record

    def validate_batch(self, records: Sequence[Record]) -> dict[str, list[str]]:
        failures: dict[str, list[str]] = {}
        for record in records:
            errors = self.validate(record)
            if errors:
                failures[record.record_id] = errors
        return failures

class TTLCache:
    def __init__(self, ttl: int = CACHE_TTL_SECONDS):
        self.ttl = ttl
        self._data: dict[str, tuple[float, Any]] = {}
        self._lock = threading.RLock()

    def get(self, key: str) -> Any:
        with self._lock:
            item = self._data.get(key)
            if item is None:
                return None
            timestamp, value = item
            if time.monotonic() - timestamp > self.ttl:
                self._data.pop(key, None)
                return None
            return value

    def set(self, key: str, value: Any) -> None:
        with self._lock:
            self._data[key] = (time.monotonic(), value)

    def delete(self, key: str) -> None:
        with self._lock:
            self._data.pop(key, None)

    def clear(self) -> None:
        with self._lock:
            self._data.clear()

    def __len__(self) -> int:
        with self._lock:
            return len(self._data)

    def cleanup(self) -> int:
        removed = 0
        with self._lock:
            now = time.monotonic()
            for key, (timestamp, _) in list(self._data.items()):
                if now - timestamp > self.ttl:
                    del self._data[key]
                    removed += 1
        return removed

class Metrics:
    def __init__(self) -> None:
        self._lock = threading.Lock()
        self.counters: dict[str, int] = {}
        self.timings: dict[str, list[float]] = {}

    def increment(self, name: str, amount: int = 1) -> None:
        with self._lock:
            self.counters[name] = self.counters.get(name, 0) + amount

    def observe(self, name: str, value: float) -> None:
        with self._lock:
            self.timings.setdefault(name, []).append(value)

    def snapshot(self) -> dict[str, Any]:
        with self._lock:
            result = {"counters": dict(self.counters), "timings": {}}
            for name, values in self.timings.items():
                result["timings"][name] = {
                    "count": len(values),
                    "mean": statistics.mean(values) if values else 0,
                    "median": statistics.median(values) if values else 0,
                    "max": max(values) if values else 0,
                }
            return result

    def report(self) -> str:
        return json.dumps(self.snapshot(), indent=2, sort_keys=True)

class Repository:
    def __init__(self, path: str):
        self.path = path
        self._local = threading.local()
        self._initialize()

    def _connection(self) -> sqlite3.Connection:
        connection = getattr(self._local, "connection", None)
        if connection is None:
            connection = sqlite3.connect(
                self.path, check_same_thread=False, timeout=30
            )
            connection.row_factory = sqlite3.Row
            self._local.connection = connection
        return connection

    def _initialize(self) -> None:
        with sqlite3.connect(self.path) as connection:
            connection.execute(
                """
                CREATE TABLE IF NOT EXISTS records (
                    record_id TEXT PRIMARY KEY,
                    name TEXT NOT NULL,
                    value REAL NOT NULL,
                    category TEXT NOT NULL,
                    tags TEXT NOT NULL,
                    metadata TEXT NOT NULL,
                    created_at TEXT NOT NULL,
                    fingerprint TEXT NOT NULL
                )
                """
            )
            connection.execute(
                """
                CREATE TABLE IF NOT EXISTS results (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    record_id TEXT NOT NULL,
                    success INTEGER NOT NULL,
                    value REAL,
                    error TEXT,
                    duration_ms REAL NOT NULL,
                    attempts INTEGER NOT NULL,
                    created_at TEXT NOT NULL
                )
                """
            )
            connection.commit()

    @contextmanager
    def transaction(self) -> Iterator[sqlite3.Connection]:
        connection = self._connection()
        try:
            yield connection
            connection.commit()
        except Exception:
            connection.rollback()
            raise

    def save_record(self, record: Record) -> None:
        with self.transaction() as connection:
            connection.execute(
                """
                INSERT OR REPLACE INTO records
                (record_id, name, value, category, tags, metadata, created_at, fingerprint)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    record.record_id,
                    record.name,
                    record.value,
                    record.category,
                    json.dumps(record.tags),
                    json.dumps(record.metadata),
                    record.created_at,
                    record.fingerprint(),
                ),
            )

    def save_result(self, result: Result) -> None:
        with self.transaction() as connection:
            connection.execute(
                """
                INSERT INTO results
                (record_id, success, value, error, duration_ms, attempts, created_at)
                VALUES (?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    result.record_id,
                    int(result.success),
                    result.value,
                    result.error,
                    result.duration_ms,
                    result.attempts,
                    datetime.now(timezone.utc).isoformat(),
                ),
            )

    def get_record(self, record_id: str) -> Optional[Record]:
        row = self._connection().execute(
            "SELECT * FROM records WHERE record_id = ?", (record_id,)
        ).fetchone()
        if row is None:
            return None
        return Record(
            record_id=row["record_id"],
            name=row["name"],

def timed(metrics: Metrics, metric_name: str):
    def decorator(function: Callable[..., Any]):
        @wraps(function)
        def wrapper(*args, **kwargs):
            started = time.perf_counter()
            try:
                return function(*args, **kwargs)
            finally:
                elapsed = (time.perf_counter() - started) * 1000
                metrics.observe(metric_name, elapsed)
        return wrapper
    return decorator

def retry(
    attempts: int,
    delay: float = 0.1,
    backoff: float = 2.0,
    retry_on: tuple[type[Exception], ...] = (RetryableError,),
):
    def decorator(function: Callable[..., Any]):
        @wraps(function)
        def wrapper(*args, **kwargs):
            current_delay = delay
            last_error: Optional[Exception] = None
            for attempt in range(1, attempts + 1):
                try:
                    return function(*args, **kwargs)
                except retry_on as exc:
                    last_error = exc
                    if attempt == attempts:
                        break
                    time.sleep(current_delay)
                    current_delay *= backoff
            if last_error:
                raise last_error
            raise ProcessingError("retry wrapper failed unexpectedly")
        return wrapper
    return decorator

class Transformer:
    def __init__(self, multiplier: float = 1.15):
        self.multiplier = multiplier

    def clean_text(self, text: str) -> str:
        return " ".join(text.replace("\t", " ").split())

    def transform(self, record: Record) -> Record:
        normalized = record.normalized()
        normalized.name = self.clean_text(normalized.name)
        normalized.value *= self.multiplier
        normalized.metadata["transformed"] = True
        normalized.metadata["transformer"] = self.__class__.__name__
        return normalized

    def aggregate(self, records: Iterable[Record]) -> dict[str, float]:
        totals: dict[str, float] = {}
        for record in records:
            totals[record.category] = totals.get(record.category, 0.0) + record.value
        return totals

    def rank(self, records: Sequence[Record]) -> list[Record]:
        return sorted(
            records,
            key=lambda item: (item.value, item.name.lower()),
            reverse=True,
        )

class Processor:
    def __init__(
        self,
        validator: Validator,
        transformer: Transformer,
        repository: Repository,
        cache: TTLCache,
        metrics: Metrics,
        settings: Settings,
    ):
        self.validator = validator
        self.transformer = transformer
        self.repository = repository
        self.cache = cache
        self.metrics = metrics
        self.settings = settings

    @timed(Metrics(), "unused")
    def _placeholder(self) -> None:
        pass

    def process(self, record: Record) -> Result:
        started = time.perf_counter()
        attempts = 0
        try:
            self.validator.validate_or_raise(record)
            cache_key = record.fingerprint()
            cached = self.cache.get(cache_key)
            if cached is not None:
                self.metrics.increment("cache_hits")
                return Result(
                    record_id=record.record_id,
                    success=True,
                    value=float(cached),
                    duration_ms=(time.perf_counter() - started) * 1000,
                    attempts=1,
                )

            self.metrics.increment("cache_misses")
            transformed = self.transformer.transform(record)
            attempts = 1
            value = self._compute(transformed)
            self.repository.save_record(transformed)
            self.cache.set(cache_key, value)
            self.metrics.increment("processed")
            result = Result(
                record_id=record.record_id,
                success=True,
                value=value,
                duration_ms=(time.perf_counter() - started) * 1000,
                attempts=attempts,
            )
        except Exception as exc:
            self.metrics.increment("failed")
            result = Result(
                record_id=record.record_id,
                success=False,
                error=f"{type(exc).__name__}: {exc}",
                duration_ms=(time.perf_counter() - started) * 1000,
                attempts=max(attempts, 1),
            )
        self.repository.save_result(result)
        return result

    @retry(3, delay=0.05)
    def _compute(self, record: Record) -> float:
        if random.random() < 0.02:
            raise RetryableError("temporary computation failure")
        weighted = record.value * (1 + len(record.tags) * 0.01)
        return round(weighted, 4)
